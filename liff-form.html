/** * LINE OA Receipt Logger — Google Apps Script backend * * Sheet layout expected (create these 4 tabs in your Google Sheet): * * 1) "Receipts" — the log. One row per bill/credit note. Header row (row 1): * Receipt ID | Date | Time | Customer | Customer ID | Salesperson | Sales ID | * Type | Billing # | Amount | Payment Type | Payment Date | Cheque Type | * Cheque Date | Receipt Image | Status | Voided At * * "Receipt ID" ties together every row that came from the same submission * (e.g. 2 bills + 1 credit note = 3 rows, same Receipt ID) — it's what lets * the "My receipts" screen group and void them together. * "Type" is either "Bill" or "Credit Note". Credit Note rows carry a * negative Amount. * "Status" is "Active" or "Voided". Voiding never deletes or overwrites a * row — it just flags it, so the original entry stays in the audit trail. * A sale can only void their own receipts (matched on Sales ID), and only * receipts from the last 7 days show up in "My receipts" at all. * * 2) "Salespersons" — header row: Sales ID | Name * (one salesperson per row below the header) * * 3) "Customers" — header row: Customer ID | Customer Name * (one customer per row below the header) * * 4) "LineUsers" — header row: LINE User ID | Sales ID * (one row per salesperson, linking their real LINE account to their * Sales ID). THIS IS THE IDENTITY-BINDING TABLE — see the block comment * above getSalesIdForLineUser() below for how it's used and why it * matters. Only an admin with edit access to the Sheet can add rows here, * so it's fully under your control — the app itself never writes to it. * * DRIVE LAYOUT: * DRIVE_FOLDER_ID/ * 2026-08/ <- created automatically, one per month * Somchai Sales/ <- created automatically, one per salesperson * Somchai Sales - 20260818-143012.jpg * * SETUP: * 1. Replace SHEET_ID below with your Google Sheet's ID * (the long string in the sheet's URL between /d/ and /edit) * 2. Replace DRIVE_FOLDER_ID below with the ID of the root Google Drive * folder that will hold the monthly receipt-photo subfolders (the * string in the folder's URL after /folders/) * 3. Run setupSheetTabs() once from the Apps Script editor to create any * missing tabs (including "LineUsers"). * 4. Add a row to "LineUsers" for each salesperson: their LINE User ID in * column A, their Sales ID (matching the "Salespersons" tab) in column B. * You can get a person's LINE User ID by having them open the LIFF form * once — it's harmless if unregistered, see notes below — or via your * LINE OA's member tools. * 5. Deploy > New deployment > type "Web app" * - Execute as: Me * - Who has access: Anyone * 6. Copy the deployment URL into GAS_ENDPOINT in liff-form.html */ const SHEET_ID = '1dYJHab1t0XkkMXI-1VW8hureTKdXhRT040N5eT8j28I'; const DRIVE_FOLDER_ID = '1_FDfi5aLkRx4rBqxTaSIMlDizkx-W1mi'; const RECEIPTS_SHEET = 'Receipts'; const SALES_SHEET = 'Salespersons'; const CUSTOMERS_SHEET = 'Customers'; const LINE_USERS_SHEET = 'LineUsers'; // 0-indexed column positions in the "Receipts" sheet — keep in sync with the // header row appended in setupSheetTabs(). const COL = { RECEIPT_ID: 0, DATE: 1, TIME: 2, CUSTOMER: 3, CUSTOMER_ID: 4, SALESPERSON: 5, SALES_ID: 6, TYPE: 7, BILL_NO: 8, AMOUNT: 9, PAYMENT_TYPE: 10, PAYMENT_DATE: 11, CHEQUE_TYPE: 12, CHEQUE_DATE: 13, RECEIPT_IMAGE: 14, STATUS: 15, VOIDED_AT: 16 }; function doGet(e) { const action = e && e.parameter ? e.parameter.action : null; if (action === 'lists') { const lineUserId = e.parameter.lineUserId || ''; return jsonResponse(getLists(lineUserId)); } if (action === 'myReceipts') { const salesId = e.parameter.salesId; if (!salesId) return jsonResponse({ error: 'Missing salesId' }); return jsonResponse({ receipts: getMyReceipts(salesId) }); } return jsonResponse({ error: 'Unknown action. Call this URL with ?action=lists or ?action=myReceipts&salesId=...' }); } function doPost(e) { try { if (!e || !e.postData) { return jsonResponse({ success: false, error: 'No POST data received' }); } const data = JSON.parse(e.postData.contents); if (data.action === 'void') { const voided = voidReceipt(data); return jsonResponse({ success: true, voided: voided }); } const identityError = checkIdentity(data); if (identityError) { return jsonResponse({ success: false, error: identityError }); } const validationError = validateReceipt(data); if (validationError) { return jsonResponse({ success: false, error: validationError }); } let imageUrl = ''; if (data.receiptImage) { try { imageUrl = uploadReceiptImage(data); } catch (imgErr) { // Don't block the receipt from being logged just because the image upload failed imageUrl = 'Upload failed: ' + imgErr.message; } } const result = saveReceipt(data, imageUrl); return jsonResponse({ success: true, receiptId: result.receiptId, rows: result, imageUrl: imageUrl }); } catch (err) { return jsonResponse({ success: false, error: err.message }); } } /** * ── IDENTITY BINDING ────────────────────────────────────────────────────── * Everything the client sends (salesId, salesperson name, etc.) is just data * in a POST body — a browser can send whatever it wants. On its own, that * means "sales can only touch their own stuff" is only a UI convention, not * a real guarantee: anyone who inspects the request could submit a * different Sales ID and the server would have no way to know it wasn't * really them. * * lineUserId is different: it comes from liff.getProfile() inside the LINE * app, and identifies the actual logged-in LINE account. It can't be typed * in or chosen — LINE itself vouches for it. * * The "LineUsers" sheet tab is the admin-controlled link between those two * worlds: LINE User ID -> Sales ID. Only someone with edit access to the * Sheet can add or change a row there, so it's fully under your control — * the web app never writes to this tab. * * getSalesIdForLineUser() looks up that mapping. checkIdentity() then makes * sure that if a mapping exists for the LINE account making the request, * the Sales ID in the request actually matches it — if someone tries to * submit or void as a different salesperson than the LINE account they're * logged in as, this rejects the request before anything is written. * * If a LINE account has NO row in "LineUsers" yet, requests are still * allowed through unlinked (so a brand-new salesperson isn't locked out * before you've had a chance to register them) — but once you add their * row, the check becomes strict for them from then on. If you'd rather * block anyone unregistered entirely, see the REQUIRE_REGISTRATION flag * below. * ──────────────────────────────────────────────────────────────────────── */ const REQUIRE_REGISTRATION = false; // set true to hard-block any LINE account with no LineUsers row function getSalesIdForLineUser(lineUserId) { if (!lineUserId) return null; const ss = SpreadsheetApp.openById(SHEET_ID); const sheet = ss.getSheetByName(LINE_USERS_SHEET); if (!sheet) return null; // tab not created yet — treat as "no mappings configured" const values = sheet.getDataRange().getValues(); for (let i = 1; i < values.length; i++) { if (String(values[i][0]).trim() === String(lineUserId).trim()) { return String(values[i][1]).trim(); } } return null; } function checkIdentity(data) { const lineUserId = data.lineUserId || ''; if (!lineUserId) { return REQUIRE_REGISTRATION ? 'Missing LINE identity — please reopen this form from LINE.' : null; // soft mode: allow through if opened outside LINE (e.g. testing) } const mappedSalesId = getSalesIdForLineUser(lineUserId); if (!mappedSalesId) { return REQUIRE_REGISTRATION ? 'Your LINE account isn\'t registered to a salesperson yet. Ask your admin to add you to the LineUsers sheet.' : null; // soft mode: unregistered accounts pass through, UI just won't auto-lock } if (String(data.salesId || '').trim() !== mappedSalesId) { return 'Your LINE account is registered as a different salesperson. If this is wrong, ask your admin to check the LineUsers sheet.'; } return null; // matched — proceed } function validateReceipt(data) { const required = ['salesperson', 'salesId', 'customerId', 'customerName', 'paymentType', 'paymentDate']; for (const key of required) { if (!data[key] && data[key] !== 0) return `Missing field: ${key}`; } if (!Array.isArray(data.bills) || data.bills.length === 0) { return 'At least one bill (or credit note) is required'; } for (const bill of data.bills) { if (!bill.billNo) return 'Every bill needs a billing number'; if (bill.amount === undefined || bill.amount === null || isNaN(parseFloat(bill.amount))) { return `Amount must be a number for bill ${bill.billNo}`; } if (bill.type !== 'Bill' && bill.type !== 'Credit Note') { return `Unknown bill type for ${bill.billNo}`; } } if (data.paymentType === 'Cheque' && !data.chequeDate) return 'Cheque date is required for cheque payments'; return null; } /** * Returns salespersons + customers, plus (if lineUserId maps to a * salesperson) which Sales ID that LINE account is registered as, so the * frontend can auto-select and lock the salesperson dropdown. */ function getLists(lineUserId) { const ss = SpreadsheetApp.openById(SHEET_ID); const salesRows = ss.getSheetByName(SALES_SHEET).getDataRange().getValues(); const salespersons = salesRows.slice(1) .filter(r => r[0]) .map(r => ({ id: String(r[0]).trim(), name: String(r[1] || '').trim() })); const custRows = ss.getSheetByName(CUSTOMERS_SHEET).getDataRange().getValues(); const customers = custRows.slice(1) .filter(r => r[0]) .map(r => ({ id: String(r[0]).trim(), name: String(r[1] || '').trim() })); const matchedSalesId = getSalesIdForLineUser(lineUserId); return { salespersons, customers, matchedSalesId: matchedSalesId || null }; } /** * Appends one row per bill/credit note in data.bills, all sharing the same * Receipt ID and receipt-level info (customer, salesperson, payment details, * image link). Returns the generated Receipt ID and the range of rows written. */ function saveReceipt(data, imageUrl) { const ss = SpreadsheetApp.openById(SHEET_ID); const sheet = ss.getSheetByName(RECEIPTS_SHEET); const now = new Date(); const tz = ss.getSpreadsheetTimeZone(); const dateStr = Utilities.formatDate(now, tz, 'yyyy-MM-dd'); const timeStr = Utilities.formatDate(now, tz, 'HH:mm:ss'); const receiptId = Utilities.getUuid(); const firstRow = sheet.getLastRow() + 1; data.bills.forEach(bill => { const row = [ receiptId, // Receipt ID (groups this submission's rows) dateStr, // Date (system entry date) timeStr, // Time (system entry time) data.customerName, // Customer data.customerId, // Customer ID data.salesperson, // Salesperson data.salesId, // Sales ID bill.type, // Type (Bill / Credit Note) bill.billNo, // Billing # parseFloat(bill.amount), // Amount (negative for Credit Note) data.paymentType, // Payment Type data.paymentDate, // Payment Date (user-entered) data.paymentType === 'Cheque' ? (data.chequeType || '') : '', // Cheque Type data.paymentType === 'Cheque' ? (data.chequeDate || '') : '', // Cheque Date imageUrl || '', // Receipt Image (Drive link) 'Active', // Status '' // Voided At ]; sheet.appendRow(row); }); return { receiptId: receiptId, first: firstRow, last: sheet.getLastRow() }; } /** * Groups this salesperson's rows from the last 7 days by Receipt ID. * Rows with no Receipt ID (logged before the schema update) are skipped * since they can't be reliably grouped or voided. */ function getMyReceipts(salesId) { const ss = SpreadsheetApp.openById(SHEET_ID); const sheet = ss.getSheetByName(RECEIPTS_SHEET); const values = sheet.getDataRange().getValues(); const tz = ss.getSpreadsheetTimeZone(); const cutoff = new Date(); cutoff.setDate(cutoff.getDate() - 7); const byReceipt = {}; const order = []; for (let i = 1; i < values.length; i++) { const r = values[i]; if (String(r[COL.SALES_ID]).trim() !== String(salesId).trim()) continue; const id = r[COL.RECEIPT_ID]; if (!id) continue; const rowDate = new Date(r[COL.DATE]); if (isNaN(rowDate) || rowDate < cutoff) continue; if (!byReceipt[id]) { byReceipt[id] = { receiptId: id, date: Utilities.formatDate(rowDate, tz, 'yyyy-MM-dd'), customerId: r[COL.CUSTOMER_ID], customerName: r[COL.CUSTOMER], paymentType: r[COL.PAYMENT_TYPE], paymentDate: r[COL.PAYMENT_DATE], chequeType: r[COL.CHEQUE_TYPE], chequeDate: r[COL.CHEQUE_DATE], receiptImage: r[COL.RECEIPT_IMAGE], status: r[COL.STATUS] || 'Active', bills: [], total: 0 }; order.push(id); } const amount = parseFloat(r[COL.AMOUNT]) || 0; byReceipt[id].bills.push({ type: r[COL.TYPE], billNo: r[COL.BILL_NO], amount: amount }); byReceipt[id].total += amount; } return order.reverse().map(id => byReceipt[id]); // most recent first } /** * Marks every row belonging to a Receipt ID as "Voided" (never deletes or * overwrites the original values). Requires BOTH: * (a) salesId matches the rows' Sales ID (existing check), AND * (b) if the requesting LINE account is registered in LineUsers, its * mapped Sales ID matches the salesId being used (checkIdentity()) — * this is what stops someone from voiding as a name that isn't * actually the LINE account they're logged in as. */ function voidReceipt(data) { if (!data.receiptId) throw new Error('Missing receiptId'); if (!data.salesId) throw new Error('Missing salesId — cannot verify ownership'); const identityError = checkIdentity(data); if (identityError) throw new Error(identityError); const ss = SpreadsheetApp.openById(SHEET_ID); const sheet = ss.getSheetByName(RECEIPTS_SHEET); const values = sheet.getDataRange().getValues(); const tz = ss.getSpreadsheetTimeZone(); const now = Utilities.formatDate(new Date(), tz, 'yyyy-MM-dd HH:mm:ss'); // First pass: find the rows for this receipt and check they all belong to // the requesting salesperson BEFORE writing anything. This avoids a // partial void (some rows voided, some not) and lets us give a precise // error ("not yours") instead of a generic "not found" when the receipt // exists but belongs to someone else. const matchingRowIndexes = []; let receiptExists = false; let ownedByRequester = true; for (let i = 1; i < values.length; i++) { if (values[i][COL.RECEIPT_ID] !== data.receiptId) continue; receiptExists = true; if (String(values[i][COL.SALES_ID]).trim() !== String(data.salesId).trim()) { ownedByRequester = false; continue; } if (values[i][COL.STATUS] === 'Voided') continue; // already voided, skip re-flagging matchingRowIndexes.push(i); } if (!receiptExists) throw new Error('Receipt not found'); if (!ownedByRequester) throw new Error('You can only void your own receipts'); if (matchingRowIndexes.length === 0) throw new Error('Receipt already voided'); matchingRowIndexes.forEach(i => { sheet.getRange(i + 1, COL.STATUS + 1).setValue('Voided'); sheet.getRange(i + 1, COL.VOIDED_AT + 1).setValue(now); }); return matchingRowIndexes.length; } /** * Decodes a base64 image sent from the form and saves it into * DRIVE_FOLDER_ID / / / - . * creating the month folder and the salesperson subfolder if they don't * already exist. Returns a shareable view link. */ function uploadReceiptImage(data) { const rootFolder = DriveApp.getFolderById(DRIVE_FOLDER_ID); // Month folder, e.g. "2026-08" — based on the receipt's payment date const monthLabel = (data.paymentDate || Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyy-MM-dd')).slice(0, 7); const monthFolder = getOrCreateChildFolder(rootFolder, monthLabel); // Salesperson subfolder inside the month folder const safeName = (data.salesperson || 'Unknown').trim(); const spFolder = getOrCreateChildFolder(monthFolder, safeName); // data.receiptImage is a data URL like "data:image/jpeg;base64,AAAA..." const match = data.receiptImage.match(/^data:(.*?);base64,(.*)$/); const contentType = match ? match[1] : 'image/jpeg'; const base64Body = match ? match[2] : data.receiptImage; const ext = (contentType.split('/')[1] || 'jpg').replace('jpeg', 'jpg'); const bytes = Utilities.base64Decode(base64Body); const timestamp = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyyMMdd-HHmmss'); const fileName = `${safeName} - ${timestamp}.${ext}`; const blob = Utilities.newBlob(bytes, contentType, fileName); const file = spFolder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); return file.getUrl(); } /** Returns the child folder with this exact name, creating it if it doesn't exist. */ function getOrCreateChildFolder(parentFolder, name) { const existing = parentFolder.getFoldersByName(name); if (existing.hasNext()) return existing.next(); return parentFolder.createFolder(name); } function jsonResponse(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } /** * Run this once manually from the Apps Script editor to create the * four tabs with correct headers if your Sheet doesn't have them yet. * Safe to re-run — it only creates tabs that don't already exist, it never * touches or overwrites an existing tab. */ function setupSheetTabs() { const ss = SpreadsheetApp.openById(SHEET_ID); if (!ss.getSheetByName(RECEIPTS_SHEET)) { const s = ss.insertSheet(RECEIPTS_SHEET); s.appendRow(['Receipt ID','Date','Time','Customer','Customer ID','Salesperson','Sales ID','Type','Billing #','Amount','Payment Type','Payment Date','Cheque Type','Cheque Date','Receipt Image','Status','Voided At']); } if (!ss.getSheetByName(SALES_SHEET)) { const s = ss.insertSheet(SALES_SHEET); s.appendRow(['Sales ID', 'Name']); } if (!ss.getSheetByName(CUSTOMERS_SHEET)) { const s = ss.insertSheet(CUSTOMERS_SHEET); s.appendRow(['Customer ID', 'Customer Name']); } if (!ss.getSheetByName(LINE_USERS_SHEET)) { const s = ss.insertSheet(LINE_USERS_SHEET); s.appendRow(['LINE User ID', 'Sales ID']); } }
